// Function to Validate if the data is an integer
FUNCTION ValidateIntegerOnly(STRING input) RETURN INTEGER
    STRING input
    INTEGER value
    WHILE TRUE 
        // Use structure binding to get the pointer of the string and the error message at once
        AUTO [pointer, errorC] = PARSE ASCII CHARACTERS FROM START ITERATOR TO END ITERATOR
                                 CONVERT THE VALID SEQUENCE TO AN INTEGER
                                 AND OVERWRITE THE 'parse_age' VARIABLE WITH THE COMPUTED RESULT
        // First guard clause: Checks if the string has other character other than numbers '2ab5c'
        IF (pointer == START ITERATOR + input SIZE) 
            // Write directly to error stream without internal buffering
            FLUSH("Syntax Error: Enter only a valid Integer number.")
            CONTINUE TO NEXT ITERATION
        ENDIF

        // 2nd guard clause: Checks if the string does not contain any number 'abc'
        IF (errorC == "Invalid argument")
            // Write directly to error stream without internal buffering
            FLUSH("Type Error: Enter integers numbers not letters, spaces, special characters")
            CONTINUE TO NEXT ITERATION
        ENDIF

        // 3rd guard clause: Checks if the string exceed the maximum valid value '9999999999999999'
        IF (errorC == "Result out of range)
            FLUSH("Overflow: You have exceeded the maximum amount.")
            CONTINUE TO NEXT ITERATION
        ENDIF

        RETURN value
    ENDWHILE
END FUNCTION
